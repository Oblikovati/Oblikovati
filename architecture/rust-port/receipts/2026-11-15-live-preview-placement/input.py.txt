import ctypes, re, subprocess, sys, time
x=ctypes.CDLL('libX11.so.6'); t=ctypes.CDLL('libXtst.so.6')
x.XOpenDisplay.argtypes=[ctypes.c_char_p]; x.XOpenDisplay.restype=ctypes.c_void_p
x.XSetInputFocus.argtypes=[ctypes.c_void_p,ctypes.c_ulong,ctypes.c_int,ctypes.c_ulong]
x.XKeysymToKeycode.argtypes=[ctypes.c_void_p,ctypes.c_ulong]; x.XKeysymToKeycode.restype=ctypes.c_ubyte
x.XFlush.argtypes=[ctypes.c_void_p]; x.XCloseDisplay.argtypes=[ctypes.c_void_p]
t.XTestFakeKeyEvent.argtypes=[ctypes.c_void_p,ctypes.c_uint,ctypes.c_int,ctypes.c_ulong]
t.XTestFakeMotionEvent.argtypes=[ctypes.c_void_p,ctypes.c_int,ctypes.c_int,ctypes.c_int,ctypes.c_ulong]
t.XTestFakeButtonEvent.argtypes=[ctypes.c_void_p,ctypes.c_uint,ctypes.c_int,ctypes.c_ulong]
d=x.XOpenDisplay(None)
if not d: raise RuntimeError('X display unavailable')
w=re.findall(r'^\s*(0x[0-9a-f]+)\s+"[^\n]*Oblikovati[^\n]*',subprocess.check_output(['xwininfo','-root','-tree'],text=True),re.M)
if len(w)!=1: raise RuntimeError(f'expected one application window, got {w!r}')
x.XSetInputFocus(d,int(w[0],16),1,0)
if sys.argv[1] in ('click','hover'):
 t.XTestFakeMotionEvent(d,-1,int(sys.argv[2]),int(sys.argv[3]),0); x.XFlush(d); time.sleep(.3)
 if sys.argv[1]=='click':
  t.XTestFakeButtonEvent(d,1,1,0); x.XFlush(d); time.sleep(.1); t.XTestFakeButtonEvent(d,1,0,0)
elif sys.argv[1] in ('ctrl-f4','alt-v'):
 sequence = [(0xffe3,1),(0xffc1,1),(0xffc1,0),(0xffe3,0)] if sys.argv[1]=='ctrl-f4' else [(0xffe9,1),(0xffe9,0),(ord('v'),1),(ord('v'),0)]
 for sym,press in sequence:
  t.XTestFakeKeyEvent(d,x.XKeysymToKeycode(d,sym),press,0); x.XFlush(d); time.sleep(.08)
else: raise ValueError(sys.argv[1])
x.XFlush(d); time.sleep(1); x.XCloseDisplay(d)

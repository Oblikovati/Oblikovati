#!/bin/bash
set -eu
printf '%s\n' "$DISPLAY" > /tmp/obk-live-save/display
printf '%s\n' "$XAUTHORITY" > /tmp/obk-live-save/xauthority
export OBK_ADDINS_DIR=/tmp/obk-live-save/addins OBK_MCP_ADDR=127.0.0.1:17952 VK_ICD_FILENAMES=/usr/share/vulkan/icd.d/lvp_icd.json XMODIFIERS=@im=none
/tmp/obk-live-save-target/debug/oblikovati > /tmp/obk-live-save/host.log 2>&1 &
host_pid=$!
printf '%s\n' "$host_pid" > /tmp/obk-live-save/pid
cleanup() { kill "$host_pid" 2>/dev/null || true; wait "$host_pid" 2>/dev/null || true; }
trap cleanup EXIT
wait "$host_pid"

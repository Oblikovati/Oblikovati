import ctypes, os, re, subprocess, sys, time
x=ctypes.CDLL('libX11.so.6'); t=ctypes.CDLL('libXtst.so.6')
x.XOpenDisplay.argtypes=[ctypes.c_char_p]; x.XOpenDisplay.restype=ctypes.c_void_p
x.XSetInputFocus.argtypes=[ctypes.c_void_p,ctypes.c_ulong,ctypes.c_int,ctypes.c_ulong]
x.XKeysymToKeycode.argtypes=[ctypes.c_void_p,ctypes.c_ulong];x.XKeysymToKeycode.restype=ctypes.c_ubyte

t.XTestFakeButtonEvent.argtypes=[ctypes.c_void_p,ctypes.c_uint,ctypes.c_int,ctypes.c_ulong]
d=x.XOpenDisplay(None)
if not d: raise RuntimeError('X display unavailable')
windows=re.findall(r'^\s*(0x[0-9a-f]+)\s+"[^\n]*Oblikovati[^\n]*',subprocess.check_output(['xwininfo','-root','-tree'],text=True),re.M)
if len(windows)!=1: raise RuntimeError(f'expected one application window, got {windows!r}')
x.XSetInputFocus(d,int(windows[0],16),1,0)
if sys.argv[1]=='resize':
 x.XMoveResizeWindow(d,int(windows[0],16),0,0,1900,1100)
elif sys.argv[1]=='move':
 t.XTestFakeMotionEvent(d,-1,int(sys.argv[2]),int(sys.argv[3]),0)
elif sys.argv[1]=='click':
 t.XTestFakeMotionEvent(d,-1,int(sys.argv[2]),int(sys.argv[3]),0);x.XFlush(d);time.sleep(0.2);t.XTestFakeButtonEvent(d,1,1,0);x.XFlush(d);time.sleep(0.2);t.XTestFakeButtonEvent(d,1,0,0)
elif sys.argv[1] in ('ctrl-n','ctrl-z','ctrl-y','ctrl-s','ctrl-w','ctrl-o','ctrl-a','ctrl-w','ctrl-o','ctrl-a'):
 key=ord(sys.argv[1][-1])
 for sym,pressed in [(0xffe3,1),(key,1),(key,0),(0xffe3,0)]: t.XTestFakeKeyEvent(d,x.XKeysymToKeycode(d,sym),pressed,0)
elif sys.argv[1] in ('tab','enter','escape'):
 sym={'tab':0xff09,'enter':0xff0d,'escape':0xff1b}[sys.argv[1]]
 code=x.XKeysymToKeycode(d,sym);t.XTestFakeKeyEvent(d,code,1,0);x.XFlush(d);time.sleep(0.1);t.XTestFakeKeyEvent(d,code,0,0)
elif sys.argv[1]=='type':
 for ch in sys.argv[2]:
  code=x.XKeysymToKeycode(d,ord(ch))
  if not code: raise ValueError(f'no keycode for {ch!r}')
  t.XTestFakeKeyEvent(d,code,1,0);x.XFlush(d);time.sleep(0.025);t.XTestFakeKeyEvent(d,code,0,0);x.XFlush(d);time.sleep(0.025)
else: raise ValueError(sys.argv[1])
x.XFlush(d);time.sleep(1);x.XCloseDisplay(d)

#!/bin/bash
set -eu
printf '%s\n' "$DISPLAY" > /tmp/obk-live-view/display
printf '%s\n' "$XAUTHORITY" > /tmp/obk-live-view/xauthority
export OBK_CONFIG_DIR=/tmp/obk-live-view/config OBK_ADDINS_DIR=/tmp/obk-live-view/addins OBK_MCP_ADDR=127.0.0.1:17954 VK_ICD_FILENAMES=/usr/share/vulkan/icd.d/lvp_icd.json XMODIFIERS=@im=none
/tmp/obk-live-view-target/debug/oblikovati > /tmp/obk-live-view/host.log 2>&1 &
host_pid=$!
printf '%s\n' "$host_pid" > /tmp/obk-live-view/pid
cleanup() { kill "$host_pid" 2>/dev/null || true; wait "$host_pid" 2>/dev/null || true; }
trap cleanup EXIT
wait "$host_pid"

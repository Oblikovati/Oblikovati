#[test]
fn published_docking_part_shell_preview_probe() {
    let mut world = ApplicationWorld::new();
    let fixture =
        super::super::scenario_root().join("fixtures/cad/docking-visible-solid-current.opd");
    world.open_fixture(&fixture);
    let shell = diagnostics::pointer_pixels(world.run("Shell").center());
    super::super::x11_input::click(shell);
    world.pump(Vec::new());
    assert!(diagnostics::window_geometry("Shell").is_some());
    let face = super::super::ribbon_keyboard_navigation::visible_face_point(&mut world);
    super::super::x11_input::click(face);
    world.pump(Vec::new());
    assert!(world.state().model_face_chosen);
    world.expect_text("Thickness");
    let status = world.status_text();
    eprintln!("Published Shell fixture {fixture:?}: {status}");
    assert!(world.text().to_lowercase().contains("tolerance"));
    world.save_capture("published-shell-refusal");
    world.close();
}
